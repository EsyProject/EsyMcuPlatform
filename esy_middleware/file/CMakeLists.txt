foreach(dependency IN ITEMS esy_common_defs esy_opt_default)
    if(NOT TARGET ${dependency})
        message(FATAL_ERROR "esy_file requires target ${dependency}")
    endif()
endforeach()
add_library(esy_file STATIC src/esy_file.c src/adapters/esy_file_ram.c)
target_compile_features(esy_file PUBLIC c_std_11)
target_include_directories(esy_file PUBLIC ${CMAKE_CURRENT_LIST_DIR}/include)
target_link_libraries(esy_file PUBLIC esy_common_defs PRIVATE esy_opt_default)
if(ENABLE_ESY_FILE_FLASH)
    if(NOT ENABLE_ESY_FLASH OR NOT TARGET esy_hal)
        message(FATAL_ERROR "ENABLE_ESY_FILE_FLASH requires ENABLE_ESY_FLASH and target esy_hal")
    endif()
    add_library(esy_file_flash STATIC src/esy_file_flash.c)
    target_link_libraries(esy_file_flash PUBLIC esy_file PRIVATE esy_hal esy_opt_default)
endif()
if(ENABLE_ESY_FILE_FS)
    if(NOT ENABLE_ESY_FS OR NOT TARGET esy_fs)
        message(FATAL_ERROR "ENABLE_ESY_FILE_FS requires ENABLE_ESY_FS and target esy_fs")
    endif()
    add_library(esy_file_fs STATIC src/adapters/esy_file_fs.c)
    target_link_libraries(esy_file_fs PUBLIC esy_file esy_fs PRIVATE esy_opt_default)
endif()
if(ENABLE_ESY_FILE_DUAL_FLASH_STORE)
    if(NOT ENABLE_ESY_DUAL_FLASH_STORE OR NOT TARGET esy_dual_flash_store)
        message(FATAL_ERROR "ENABLE_ESY_FILE_DUAL_FLASH_STORE requires ENABLE_ESY_DUAL_FLASH_STORE and target esy_dual_flash_store")
    endif()
    add_library(esy_file_dual_flash_store STATIC src/adapters/esy_file_dual_flash_store.c)
    target_link_libraries(esy_file_dual_flash_store PUBLIC esy_file esy_dual_flash_store PRIVATE esy_opt_default)
endif()
