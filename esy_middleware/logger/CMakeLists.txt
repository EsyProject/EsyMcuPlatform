foreach(dependency IN ITEMS esy_common_defs esy_opt_default)
    if(NOT TARGET ${dependency})
        message(FATAL_ERROR "esy_logger_core requires target ${dependency}")
    endif()
endforeach()

add_library(esy_logger_core STATIC)
target_compile_features(esy_logger_core PUBLIC c_std_11)
target_include_directories(esy_logger_core PUBLIC ${CMAKE_CURRENT_LIST_DIR}/include)
target_link_libraries(esy_logger_core PUBLIC esy_common_defs PRIVATE esy_opt_default)

add_library(esy_logger INTERFACE)
target_link_libraries(esy_logger INTERFACE esy_logger_core)

foreach(backend IN ITEMS uart printf rtt)
    string(TOUPPER "${backend}" backend_upper)
    if(NOT ENABLE_ESY_LOGGER_BACKEND_${backend_upper})
        continue()
    endif()
    if(NOT ESY_RTOS STREQUAL "None" AND NOT ESY_RTOS STREQUAL "FreeRTOS")
        message(FATAL_ERROR "Logger backends require ESY_RTOS=None or ESY_RTOS=FreeRTOS")
    endif()
    if(NOT ENABLE_ESY_SYSTEM)
        message(FATAL_ERROR "The ${backend} logger backend requires ENABLE_ESY_SYSTEM")
    endif()
    if(NOT TARGET esy_hal)
        message(FATAL_ERROR "The ${backend} logger backend requires target esy_hal")
    endif()
    if(backend STREQUAL "uart" AND (NOT ENABLE_ESY_UART OR NOT ENABLE_ESY_DMA))
        message(FATAL_ERROR "The UART logger backend requires ENABLE_ESY_UART and ENABLE_ESY_DMA, including when configured for short sends.")
    endif()
    if(NOT backend STREQUAL "rtt")
        if(NOT TARGET esy_common_ds)
            message(FATAL_ERROR "Buffered logger backends require ENABLE_ESY_COMMON_DS and esy_common_ds.")
        endif()
        if(NOT TARGET esy_common_utils)
            message(FATAL_ERROR "Buffered logger backends require target esy_common_utils")
        endif()
    endif()

    set(core esy_logger_backend_${backend}_core)
    add_library(${core} STATIC)
    target_link_libraries(${core} PUBLIC esy_logger_core PRIVATE esy_hal esy_opt_default)
    if(NOT backend STREQUAL "rtt")
        target_link_libraries(${core} PRIVATE esy_common_ds esy_common_utils)
    endif()

    if(ESY_RTOS STREQUAL "FreeRTOS")
        if(NOT TARGET esy_os)
            message(FATAL_ERROR "Logger OS adapters require the esy_os target.")
        endif()
        set(adapter esy_logger_backend_${backend})
        add_library(${adapter} STATIC)
        target_link_libraries(${adapter} PUBLIC ${core} PRIVATE esy_hal esy_opt_default)
        if(backend STREQUAL "rtt")
            target_link_libraries(${adapter} PRIVATE esy_os)
        else()
            target_link_libraries(${adapter} PUBLIC esy_os)
        endif()
        target_link_libraries(esy_logger INTERFACE ${adapter})
    else()
        target_link_libraries(esy_logger INTERFACE ${core})
    endif()
endforeach()

if(ENABLE_ESY_LOGGER_BACKEND_RTT)
    if(DEFINED ESY_SEGGER_RTT_TARGET AND TARGET ${ESY_SEGGER_RTT_TARGET})
        target_link_libraries(esy_logger_backend_rtt_core PRIVATE ${ESY_SEGGER_RTT_TARGET})
    elseif(TARGET SEGGER_RTT)
        target_link_libraries(esy_logger_backend_rtt_core PRIVATE SEGGER_RTT)
    elseif(DEFINED ESY_SEGGER_RTT_ROOT)
        target_sources(esy_logger_backend_rtt_core PRIVATE ${ESY_SEGGER_RTT_ROOT}/RTT/SEGGER_RTT.c)
        target_include_directories(esy_logger_backend_rtt_core PRIVATE
            ${ESY_SEGGER_RTT_ROOT}/RTT ${ESY_SEGGER_RTT_ROOT}/Config)
        set_source_files_properties(${ESY_SEGGER_RTT_ROOT}/RTT/SEGGER_RTT.c
            PROPERTIES COMPILE_OPTIONS "-Wno-cast-qual;-Wno-cast-align")
    else()
        message(FATAL_ERROR
            "ENABLE_ESY_LOGGER_BACKEND_RTT requires SEGGER RTT. "
            "Set ESY_SEGGER_RTT_TARGET, provide a SEGGER_RTT target, "
            "or set ESY_SEGGER_RTT_ROOT.")
    endif()
endif()
