idf_build_get_property(ESY_NATIVE_PROFILE ESY_NATIVE_PROFILE)
set(native_components freertos esp_system esp_hw_support xtensa esp_timer)
if(NOT ESY_NATIVE_PROFILE STREQUAL "core")
    list(APPEND native_components esp_driver_gpio esp_driver_uart esp_driver_i2c
        esp_driver_spi esp_driver_ledc esp_driver_mcpwm spi_flash esp_adc driver)
endif()
if(ESY_NATIVE_PROFILE STREQUAL "managed")
    list(APPEND native_components app_update esp_partition)
endif()
if(ESY_NATIVE_PROFILE STREQUAL "radio")
    list(APPEND native_components esp_wifi esp_netif esp_event lwip)
endif()
if(ESY_NATIVE_PROFILE STREQUAL "power")
    list(APPEND native_components esp_pm)
endif()
idf_component_register(SRCS main.c core.c REQUIRES ${native_components})
if(CMAKE_BUILD_EARLY_EXPANSION)
    return()
endif()
if(ESY_NATIVE_PROFILE STREQUAL "radio")
    target_sources(${COMPONENT_LIB} PRIVATE radio.c)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE ESY_TEST_NATIVE_RADIO)
    target_link_libraries(${COMPONENT_LIB} PRIVATE esy_wifi_manager esy_network_lwip)
endif()
add_subdirectory("${ESY_NATIVE_PLATFORM_ROOT}" "${CMAKE_BINARY_DIR}/esy_platform")
target_link_libraries(${COMPONENT_LIB} PRIVATE esy_hal esy_os)
if(NOT ESY_NATIVE_PROFILE STREQUAL "core")
    target_sources(${COMPONENT_LIB} PRIVATE peripherals.c)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE ESY_TEST_NATIVE_PERIPHERALS)
endif()
if(ESY_NATIVE_PROFILE STREQUAL "managed")
    target_sources(${COMPONENT_LIB} PRIVATE managed.c)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE ESY_TEST_NATIVE_MANAGED)
    target_link_libraries(${COMPONENT_LIB} PRIVATE esy_bootloader esy_ota_flash esy_coredump_core
        esy_spi_scheduler esy_uart_tx_scheduler_os)
endif()
if(ESY_NATIVE_PROFILE STREQUAL "power")
    target_sources(${COMPONENT_LIB} PRIVATE power.c)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE ESY_TEST_NATIVE_POWER)
endif()
target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror)
foreach(native_target esy_hal esy_hal_backend esy_os esy_os_context)
    target_compile_options(${native_target} PRIVATE -Wall -Wextra -Werror)
endforeach()
